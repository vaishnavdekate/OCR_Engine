cmake_minimum_required(VERSION 3.20)
project(receipt_ocr LANGUAGES C CXX)
set(PPOCR_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(PPOCR_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(PPOCR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PPOCR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PPOCR_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(PPOCR_ENABLE_METAL OFF CACHE BOOL "" FORCE)
add_subdirectory(rapidocrlite)
add_executable(receipt-ocr receipt_ocr.cpp)
target_compile_features(receipt-ocr PRIVATE cxx_std_20)
target_link_libraries(receipt-ocr PRIVATE RapidOCRLite::static)
set_target_properties(receipt-ocr PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
